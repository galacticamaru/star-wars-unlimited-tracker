UPDATE "trade_manual_wants" tmw
SET "card_printing_id" = cp.id
FROM "card_printings" cp
WHERE cp.card_definition_id = tmw.card_definition_id

--> statement-breakpoint

const updateVariant = async (cardPrintingId: number, newCount: number) => {
  if (!isAuthenticated) {
    router.push('/login');
    return;
  }

  const val = Math.max(0, newCount);
  const prev = counts[cardPrintingId] ?? 0;

  // Optimistic update
  setCounts(prev => ({ ...prev, [cardPrintingId]: val }));

  try {
    const res = await fetch('/api/collection/variants', {  // ← change to '/api/trade'
      method: 'POST',                                       // ← change to 'PATCH'
      headers: { 'Content-Type': 'application/json' },

    });
    if (!res.ok) {
      setCounts(c => ({ ...c, [cardPrintingId]: prev }));  // rollback
      console.error('Failed to update trade quantity:', await res.text());
    }
  } catch (err) {
    setCounts(c => ({ ...c, [cardPrintingId]: prev }));    // rollback on network failure
    console.error('Failed to update trade quantity:', err);
  }
};
