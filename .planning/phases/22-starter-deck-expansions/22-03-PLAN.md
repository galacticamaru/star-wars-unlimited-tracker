    ```sql
    SELECT cd.name, cd.subtitle, cp.collector_number, cp.set_code
    FROM card_printings cp
    JOIN card_definitions cd ON cp.card_definition_id = cd.id
    WHERE cd.name IN (
      'Jabba''s Guard',
      'Skiff Cargo Hold',
      'Underworld Connections',
      'Cunning Deal',
      'Payoff',
      'Bargaining for Life',
      'Boushh''s Thermals',
      'Commanding Presence',
      'Infiltration Plan'
    )
    AND cp.variant_type = 'Normal'
    ORDER BY cd.name;
    ```

    If that query returns no rows or partial results (card names may differ slightly), run the fallback query:

    ```sql
    SELECT cd.name, cd.subtitle, cp.collector_number, cp.set_code
    FROM card_printings cp
    JOIN card_definitions cd ON cp.card_definition_id = cd.id
    WHERE cp.set_code IN ('LAW')
    AND cp.variant_type = 'Normal'
    AND (
      cd.name ILIKE '%guard%' OR
      cd.name ILIKE '%skiff%' OR
      cd.name ILIKE '%connections%' OR
      cd.name ILIKE '%cunning%' OR
      cd.name ILIKE '%payoff%' OR
      cd.name ILIKE '%bargaining%' OR
      cd.name ILIKE '%boushh%' OR
      cd.name ILIKE '%commanding%' OR
      cd.name ILIKE '%infiltration%'
    )
    ORDER BY cd.name;
    ```
